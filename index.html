/* ── 3. 트리거 : 발주·입고를 이벤트로 자동 기록 ─────────────── 이미 입고 이벤트(수동·QR 포함)가 있으면 자동 생성하지 않는다. → RPC 분할입고와 이중 집계되지 않는다. */ create or replace function public.fn_outsourcing_move_sync() returns trigger language plpgsql security definer set search_path = 'public' as $$ declare v_out numeric := coalesce(nullif(new.order_qty, 0), 1); end if; elsif new.status = '발주' then delete from public.outsourcing_moves where line_id = new.line_id and io = '입고' and source = '자동'; end if; return new; end $$; drop trigger if exists trg_outsourcing_move_sync on public.order_lines; create trigger trg_outsourcing_move_sync after insert or update on public.order_lines for each row execute function public.fn_outsourcing_move_sync(); /* ── 4. 분할입고 RPC ────────────────────────────────────────── */ create or replace function public.fn_osp_receive( p_line_id bigint, p_qty numeric, p_date date default current_date, p_short numeric default 0, p_reason text default null, p_worker text default null, p_source text default '수동', p_remark text default null ) returns jsonb language plpgsql security definer set search_path = 'public' as $$ declare l public.order_lines%rowtype; end if; if coalesce(l.category, '') <> '외주가공' then raise exception '외주가공 발주라인이 아닙니다. v_new := v_got + p_qty + coalesce(p_short, 0); if v_new > v_ord then raise exception '누적 입고 %가 발주수량 %를 초과합니다.', v_new, v_ord; end if; insert into public.outsourcing_moves (line_id, job_no, part_no, machining_process_code, vendor_name, io, move_date, in_qty, short_qty, reason, remark, worker, source) values (p_line_id, l.job_no, l.part_no, l.machining_process_code, l.vendor_name, '입고', coalesce(p_date, current_date), p_qty, coalesce(p_short, 0), p_reason, p_remark, p_worker, coalesce(p_source, '수동')); v_closed := v_new >= v_ord; update public.order_lines set receipt_qty = v_new, receipt_date = case when v_closed then coalesce(p_date, current_date) else null end, status = case when v_closed then '입고' else '발주' end, updated_at = now() where line_id = p_line_id; return jsonb_build_object( 'closed', v_closed, 'in_qty', v_new, 'order_qty', v_ord, 'open_qty', greatest(v_ord - v_new, 0)); end $$; /* ── 5. 입고취소 RPC (회차 단위) ────────────────────────────── */ create or replace function public.fn_osp_receive_cancel( p_line_id bigint, p_move_id bigint default null ) returns jsonb language plpgsql security definer set search_path = 'public' as $$ declare v_ord numeric; v_new numeric; v_del bigint; begin select coalesce(nullif(order_qty, 0), 1) into v_ord from public.order_lines where line_id = p_line_id; if not found then raise exception '발주라인 %를 찾을 수 없습니다.', p_line_id; end if; select move_id into v_del from public.outsourcing_moves where line_id = p_line_id and io = '입고' and (p_move_id is null or move_id = p_move_id) order by move_date desc, move_id desc limit 1; if v_del is null then raise exception '취소할 입고 기록이 없습니다.'; end if; delete from public.outsourcing_moves where move_id = v_del; select coalesce(sum(coalesce(in_qty, 0) + coalesce(short_qty, 0)), 0) into v_new from public.outsourcing_moves where line_id = p_line_id and io in ('입고', '사내입고'); update public.order_lines set receipt_qty = nullif(v_new, 0), receipt_date = null, status = '발주', updated_at = now() where line_id = p_line_id; return jsonb_build_object('canceled_move', v_del, 'in_qty', v_new, 'open_qty', greatest(v_ord - v_new, 0)); end $$; grant execute on function public.fn_osp_receive(bigint, numeric, date, numeric, text, text, text, text) to authenticated, service_role; grant execute on function public.fn_osp_receive_cancel(bigint, bigint) to authenticated, service_role; /* ── 6. 보유처·잔량 뷰 ──────────────────────────────────────── */ create or replace view public.v_outsourcing_position as with agg as ( select l.line_id, l.job_no, l.item_name, l.part_no, l.part_name, l.machining_process_code, l.machining_process_name, l.vendor_name, l.order_date, l.required_date, l.receipt_date, l.status, coalesce(nullif(l.order_qty, 0), 1) as order_qty, coalesce(sum(m.in_qty) filter (where m.io in ('입고','사내입고')), 0) as in_qty, coalesce(sum(m.short_qty) filter (where m.io in ('도착','입고')), 0) as short_qty, max(m.move_date) filter (where m.io = '도착') as arrive_date, count(m.move_id) filter (where m.io in ('입고','사내입고')) as receipt_cnt, count(m.move_id) as move_cnt from public.order_lines l left join public.outsourcing_moves m on m.line_id = l.line_id where l.category = '외주가공' group by l.line_id ) select a.*, greatest(a.order_qty - a.in_qty - a.short_qty, 0) as open_qty, case when greatest(a.order_qty - a.in_qty - a.short_qty, 0) <= 0 then '사내 (복귀완료)' when a.in_qty > 0 then a.vendor_name || ' (부분보유)' when a.arrive_date is not null then a.vendor_name || ' (작업중)' else coalesce(a.vendor_name, '-') || ' (반출중)' end as holder, case when greatest(a.order_qty - a.in_qty - a.short_qty, 0) <= 0 then '완료' when a.in_qty > 0 then '부분완료' when a.arrive_date is not null then '작업중' else '반출중' end as position_state, case when greatest(a.order_qty - a.in_qty - a.short_qty, 0) <= 0 then null else current_date - a.order_date end as stay_days, (greatest(a.order_qty - a.in_qty - a.short_qty, 0) > 0 and a.required_date is not null and a.required_date < current_date) as delayed from agg a; grant select on public.v_outsourcing_position to anon, authenticated, service_role; comment on view public.v_outsourcing_position is '외주가공 라인별 현재 보유처·잔량·체류일. 가공구매 진척현황 / 외주가공 입고 화면에서 참조.'; /* ── 7. 기존 자료 백필 ──────────────────────────────────────── */ insert into public.outsourcing_moves (line_id, job_no, part_no, machining_process_code, vendor_name, io, move_date, out_qty, worker, source, remark) select l.line_id, l.job_no, l.part_no, l.machining_process_code, l.vendor_name, '출고', l.order_date, coalesce(nullif(l.order_qty,0),1), l.owner_name, '자동', '기존 발주 소급' from public.order_lines l where l.category = '외주가공' and l.order_date is not null on conflict do nothing; insert into public.outsourcing_moves (line_id, job_no, part_no, machining_process_code, vendor_name, io, move_date, in_qty, worker, source, remark) select l.line_id, l.job_no, l.part_no, l.machining_process_code, l.vendor_name, '입고', l.receipt_date, coalesce(nullif(l.receipt_qty,0), nullif(l.order_qty,0), 1), l.owner_name, '자동', '기존 입고 소급' from public.order_lines l where l.category = '외주가공' and l.status in ('입고','입고확정') and l.receipt_date is not null on conflict do nothing; /* ── 8. 검증 ────────────────────────────────────────────────── */ select position_state, count(*) as 건수, sum(order_qty) as 발주수량, sum(open_qty) as 미회수 from public.v_outsourcing_position group by position_state order by 1;
